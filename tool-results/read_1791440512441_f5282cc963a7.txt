     1→     1→//! Segmented multi-connection resumable download engine (v2).
     2→     2→//!
     3→     3→//! Data-integrity rules (they outrank everything else):
     4→     4→//!  - the per-segment journal NEVER leads the file: `written` is persisted after the bytes
     5→     5→//!    reach the OS, so a crash can only cost a small re-download, never a hole;
     6→     6→//!  - resume requests carry `If-Range` with the validator of the source the part came from;
     7→     7→//!    a 200 answer means the source changed → the whole run restarts from zero (bounded);
     8→     8→//!  - `206` answers are validated against the requested offset and total before any write;
     9→     9→//!  - writes are hard-capped at the planned window end — server surplus bytes are dropped;
    10→    10→//!  - completion verifies Σ written == total, then hashes the part file with SHA-256 and —
    11→    11→//!    when the job carries an expected checksum — refuses to declare success on mismatch;
    12→    12→//!  - the final rename happens only after fsync.
    13→    13→//!
    14→    14→//! Jobs whose source does not support ranges (or whose size is unknown) run as a single
    15→    15→//! append-mode window; the model is identical, only the window is open-ended. A v1-era part
    16→    16→//! (bytes on disk, validators persisted, no journal rows) is migrated forward instead of
    17→    17→//! being discarded.

   100→   100→impl EngineError {
   101→   101→    fn to_contract(&self) -> ContractError {
   102→   102→        match self {
   103→   103→            EngineError::Contract(c) => c.clone(),
   104→   104→            EngineError::Http(e) => {
   105→   105→                let status = e.status().map(|s| s.as_u16());
   106→   106→                let code = if e.is_timeout() || e.is_connect() {
   107→   107→                    ErrorCode::NetworkError
   108→   108→                } else if status.is_some() {
   109→   109→                    match status {
   110→   110→                        Some(416) => ErrorCode::RangeNotSatisfiable,
   111→   111→                        _ => ErrorCode::HttpError,
   112→   112→                    }
   113→   113→                } else {
   114→   114→                    ErrorCode::NetworkError
   115→   115→                };
   116→   116→                ContractError::new(code, format!("{e}"))
   117→   117→                    .with_data(serde_json::json!({ "status": status }))
   118→   118→            }
   119→   119→            EngineError::Io(e) => {
   120→   120→                let code = if e.kind() == std::io::ErrorKind::StorageFull {
   121→   121→                    ErrorCode::DiskFull
   122→   122→                } else {
   123→   123→                    ErrorCode::IoError
   124→   124→                };
   125→   125→                ContractError::new(code, format!("{e}"))
   126→   126→            }
   127→   127→        }
   128→   128→    }
   129→   129→}

   224→   224→    /// Create + persist a queued job; the destination dir is authoritative from creation.
   225→   225→    pub async fn create_job(&self, params: CreateJobParams) -> Result<DownloadJob, ContractError> {
   226→   226→        if !params.url.starts_with("http://") && !params.url.starts_with("https://") {
   227→   227→            return Err(ContractError::new(
   228→   228→                ErrorCode::UrlInvalid,
   229→   229→                "only http(s) URLs are supported",
   230→   230→            ));
   231→   231→        }
   232→   232→        let checksum = match &params.checksum {
   233→   233→            Some(spec) => {
   234→   234→                let value = normalize_expected(spec.algorithm, &spec.value)
   235→   235→                    .map_err(|msg| ContractError::new(ErrorCode::InvalidParams, msg))?;
   236→   236→                Some((spec.algorithm, value))
   237→   237→            }
   238→   238→            None => None,
   239→   239→        };
   240→   240→        let ts = now_ms();
   241→   241→        let job = DownloadJob {
   242→   242→            id: new_ulid(),
   243→   243→            url: params.url,
   244→   244→            destination_dir: params.destination_dir,
   245→   245→            destination_path: None, // resolved by the probe
   246→   246→            filename: params.filename,
   247→   247→            status: "queued".into(),
   248→   248→            total_bytes: None,
   249→   249→            received_bytes: 0,
   250→   250→            etag: None,
   251→   251→            last_modified: None,
   252→   252→            content_type: None,
   253→   253→            speed_bps: None,
   254→   254→            eta_seconds: None,
   255→   255→            error_code: None,
   256→   256→            headers: params.headers.clone(),
   257→   257→            segment_count: None,
   258→   258→            checksum_algorithm: checksum.as_ref().map(|(a, _)| *a),
   259→   259→            checksum_expected: checksum.as_ref().map(|(_, v)| v.clone()),
   260→   260→            checksum_actual: None,
   261→   261→            checksum_verified: None,
   262→   262→            created_at: ts,
   263→   263→            updated_at: ts,
   264→   264→        };
   265→   265→        self.store.insert_job(&job)?;
   266→   266→        Ok(job)
   267→   267→    }

   269→   269→    /// The per-job task: probe → segmented download → verify. Runs until terminal/paused.
   270→   270→    pub async fn run_job(
   271→   271→        self: Arc<Self>,
   272→   272→        mut job: DownloadJob,
   273→   273→        mut cmd_rx: mpsc::Receiver<JobCommand>,
   274→   274→        stop: CancellationToken,
   275→   275→    ) {
   276→   276→        match self.drive(&mut job, &mut cmd_rx, &stop).await {
   277→   277→            Ok(()) => {}
   278→   278→            Err(err) => {
   279→   279→                if !stop.is_cancelled() {
   280→   280→                    let cerr = err.to_contract();
   281→   281→                    tracing::warn!(job = %job.id, error = %cerr, "job failed");
   282→   282→                    set_error(&mut job, &cerr);
   283→   283→                    if self.store.update_job(&job).is_ok() {
   284→   284→                        self.emit_job(&job);
   285→   285→                    }
   286→   286→                }
   287→   287→            }
   288→   288→        }
   289→   289→    }
   290→   290→
   291→   291→    async fn drive(
   292→   292→        &self,
   293→   293→        job: &mut DownloadJob,
   294→   294→        cmd_rx: &mut mpsc::Receiver<JobCommand>,
   295→   295→        stop: &CancellationToken,
   296→   296→    ) -> Result<(), EngineError> {
   297→   297→        // Resume safety: If-Range must carry the validators of the SOURCE THE PART CAME FROM
   298→   298→        // (persisted before this run), never the freshly probed ones — otherwise a changed
   299→   299→        // source would answer 206 against the new validator and the part would be spliced
   300→   300→        // with foreign bytes. Captured BEFORE the probe overwrites job.etag/last_modified.
   301→   301→        let mut part_etag = job.etag.clone();
   302→   302→        let mut part_last_modified = job.last_modified.clone();
   303→   303→
   304→   304→        // ---- probing ----
   305→   305→        job.status = "probing".into();
   306→   306→        touch(job);
   307→   307→        self.store.update_job(job)?;
   308→   308→        self.emit_job(job);
   309→   309→
   310→   310→        let mut probe = tokio::select! {
   311→   311→            _ = stop.cancelled() => return Ok(()),
   312→   312→            r = self.probe(job) => r?,
   313→   313→        };
   314→   314→
   315→   315→        // ---- destination resolution (once; the part file is bound to this path) ----
   316→   316→        let base_name = job
   317→   317→            .filename
   318→   318→            .clone()
   319→   319→            .or_else(|| probe.filename_hint.clone())
   320→   320→            .or_else(|| filename_from_url(&job.url))
   321→   321→            .unwrap_or_else(|| "download.bin".into());
   322→   322→        let dir = PathBuf::from(&job.destination_dir);
   323→   323→        tokio::fs::create_dir_all(&dir)
   324→   324→            .await
   325→   325→            .map_err(EngineError::Io)?;
   326→   326→        let candidate = dir.join(sanitize_filename(&base_name));
   327→   327→        let final_path = if tokio::fs::try_exists(&candidate).await.unwrap_or(false) {
   328→   328→            uniquify(&candidate).await
   329→   329→        } else {
   330→   330→            candidate
   331→   331→        };
   332→   332→        let part_path = part_file(&final_path);
   333→   333→
   334→   334→        job.destination_path = Some(final_path.display().to_string());
   335→   335→        job.total_bytes = probe.total_bytes;
   336→   336→        job.etag = probe.etag.clone();
   337→   337→        job.last_modified = probe.last_modified.clone();
   338→   338→        job.content_type = probe.content_type.clone();
   339→   339→        touch(job);
   340→   340→        self.store.update_job(job)?;
   341→   341→        self.emit_job(job);
   342→   342→
   343→   343→        // ---- segmented run loop (bounded clean restarts on source change) ----
   344→   344→        let mut restarts_used = 0usize;
   345→   345→        loop {
   346→   346→            let max_conn = self.max_connections.load(Ordering::SeqCst) as usize;
   347→   347→            let windows = plan_windows(
   348→   348→                job.total_bytes,
   349→   349→                probe.ranges_supported,
   350→   350→                max_conn,
   351→   351→                MIN_SEGMENT_BYTES,
   352→   352→            );
   353→   353→
   354→   354→            // Journal reconciliation. A v1-era part (no journal rows) is migrated forward
   355→   355→            // instead of thrown away; a plan change resets the journal.
   356→   356→            let mut journal = self.store.list_segments(job.id.as_str())?;
   357→   357→            if journal.is_empty() {
   358→   358→                journal = self.migrate_v1_part(job, &part_path, &windows).await?;
   359→   359→            }
   360→   360→            if !journal_matches_plan(&journal, &windows) {
   361→   361→                tracing::info!(job = %job.id, "plan changed — resetting segment journal");
   362→   362→                journal = journal_from_plan(&windows);
   363→   363→                self.store
   364→   364→                    .replace_segments(job.id.as_str(), &journal)
   365→   365→                    .map_err(EngineError::Contract)?;
   366→   366→            }
   367→   367→            // A part with bytes but no validators cannot be proven to match the source:
   368→   368→            // integrity policy discards it rather than risking a hybrid file.
   369→   369→            let part_validators = part_etag.clone().or_else(|| part_last_modified.clone());
   370→   370→            let has_written = journal.iter().any(|s| s.written_bytes > 0);
   371→   371→            if has_written && part_validators.is_none() {
   372→   372→                tracing::warn!(job = %job.id, "unprovable part (no validators) — discarding");
   373→   373→                journal = journal_from_plan(&windows);
   374→   374→                self.store
   375→   375→                    .replace_segments(job.id.as_str(), &journal)
   376→   376→                    .map_err(EngineError::Contract)?;
   377→   377→            }
   378→   378→
   379→   379→            let offset_mode = windows.iter().all(|w| w.end.is_some());
   380→   380→            let initial_written: u64 = journal.iter().map(|s| s.written_bytes).sum();
   381→   381→
   382→   382→            // Align the part file with the journal before any writer opens it.
   383→   383→            if offset_mode {
   384→   384→                // Sparse pre-size: the windows tile [0, total) exactly.
   385→   385→                let file = OpenOptions::new()
   386→   386→                    .create(true)
   387→   387→                    .truncate(false)
   388→   388→                    .write(true)
   389→   389→                    .open(&part_path)
   390→   390→                    .await
   391→   391→                    .map_err(EngineError::Io)?;
   392→   392→                file.set_len(job.total_bytes.unwrap_or(0))
   393→   393→                    .await
   394→   394→                    .map_err(EngineError::Io)?;
   395→   395→            } else {
   396→   396→                // Append mode: truncate to the durable cursor (drops any stale crash tail).
   397→   397→                let file = OpenOptions::new()
   398→   398→                    .create(true)
   399→   399→                    .truncate(false)
   400→   400→                    .write(true)
   401→   401→                    .open(&part_path)
   402→   402→                    .await
   403→   403→                    .map_err(EngineError::Io)?;
   404→   404→                file.set_len(initial_written)
   405→   405→                    .await
   406→   406→                    .map_err(EngineError::Io)?;
   407→   407→            }
   408→   408→
   409→   409→            // The If-Range validator for every request of this run: the PART's validators
   410→   410→            // when resuming written data, otherwise the freshly probed ones.
   411→   411→            let if_range = if has_written {
   412→   412→                part_validators.clone()
   413→   413→            } else {
   414→   414→                job.etag.clone().or_else(|| job.last_modified.clone())
   415→   415→            };
   416→   416→
   417→   417→            job.segment_count = Some(windows.len().max(1) as u64);
   418→   418→            job.status = "downloading".into();
   419→   419→            job.received_bytes = initial_written;
   420→   420→            job.speed_bps = None;
   421→   421→            touch(job);
   422→   422→            self.store.update_job(job)?;
   423→   423→            self.emit_job(job);
   424→   424→
   425→   425→            let state = Arc::new(RunState {
   426→   426→                received: Arc::new(AtomicU64::new(initial_written)),
   427→   427→                restart: Arc::new(AtomicBool::new(false)),
   428→   428→                cursors: Arc::new(Mutex::new(
   429→   429→                    journal.iter().map(|s| s.written_bytes).collect(),
   430→   430→                )),
   431→   431→            });
   432→   432→
   433→   433→            // Progress ticker: one throttled stream of updates for the whole run.
   434→   434→            let ticker_stop = CancellationToken::new();
   435→   435→            let ticker = self.spawn_ticker(
   436→   436→                job.id.as_str().to_string(),
   437→   437→                job.total_bytes,
   438→   438→                state.received.clone(),
   439→   439→                ticker_stop.clone(),
   440→   440→            );
   441→   441→
   442→   442→            // One task per window.
   443→   443→            let mut tasks: JoinSet<Result<SegmentOutcome, EngineError>> = JoinSet::new();
   444→   444→            for (idx, window) in windows.iter().enumerate() {
   445→   445→                let row = journal[idx].clone();
   446→   446→                tasks.spawn(Self::run_segment(
   447→   447→                    self.store.clone(),
   448→   448→                    self.client.clone(),
   449→   449→                    job.clone(),
   450→   450→                    idx,
   451→   451→                    *window,
   452→   452→                    row,
   453→   453→                    part_path.clone(),
   454→   454→                    if_range.clone(),
   455→   455→                    state.clone(),
   456→   456→                    stop.clone(),
   457→   457→                ));
   458→   458→            }
   459→   459→
   460→   460→            let mut run_result: Result<(), EngineError> = Ok(());
   461→   461→            let mut source_changed = false;

   575→   575→

   578→   578→        }
   579→   579→    }

   600→   600→    /// v1 → v2 journal migration: a legacy single-stream part (bytes on disk, validators
   601→   601→    /// persisted, no journal rows) becomes one window covering the file so the bytes are
   602→   602→    /// kept instead of re-downloaded.
   603→   603→    async fn migrate_v1_part(
   604→   604→        &self,
   605→   605→        job: &DownloadJob,
   606→   606→        part_path: &Path,
   607→   607→        windows: &[Window],
   608→   608→    ) -> Result<Vec<SegmentRow>, EngineError> {
   609→   609→        let part_len = match tokio::fs::metadata(part_path).await {
   610→   610→            Ok(meta) => meta.len(),
   611→   611→            Err(_) => 0,
   612→   612→        };
   613→   613→        let has_validators = job.etag.is_some() || job.last_modified.is_some();
   614→   614→        let first_window_covers_start = windows.first().map(|w| w.start).unwrap_or(1) == 0;
   615→   615→
   616→   616→        if part_len == 0 || !has_validators || !first_window_covers_start {
   617→   617→            let fresh = journal_from_plan(windows);
   618→   618→            self.store
   619→   619→                .replace_segments(job.id.as_str(), &fresh)
   620→   620→                .map_err(EngineError::Contract)?;
   621→   621→            return Ok(fresh);
   622→   622→        }
   623→   623→
   624→   624→        let mut rows = journal_from_plan(windows);
   625→   625→        let window_len = rows[0].end_offset.map(|e| e - rows[0].start_offset + 1);
   626→   626→        let written = match window_len {
   627→   627→            Some(len) => part_len.min(len),
   628→   628→            None => part_len, // open-ended window: keep everything the file has
   629→   629→        };
   630→   630→        rows[0].written_bytes = written;
   631→   631→        rows[0].status = if written > 0 {
   632→   632→            "downloading".into()
   633→   633→        } else {
   634→   634→            "pending".into()
   635→   635→        };
   636→   636→        if written > 0 {
   637→   637→            tracing::info!(job = %job.id, written, "migrated v1 part into segment journal");
   638→   638→        }
   639→   639→        self.store
   640→   640→            .replace_segments(job.id.as_str(), &rows)
   641→   641→            .map_err(EngineError::Contract)?;
   642→   642→        Ok(rows)
   643→   643→    }
