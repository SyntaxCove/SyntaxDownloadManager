     1→//! Segmented multi-connection resumable download engine (v2).
     2→//!
     3→//! Data-integrity rules (they outrank everything else):
     4→//!  - the per-segment journal NEVER leads the file: `written` is persisted after the bytes
     5→//!    reach the OS, so a crash can only cost a small re-download, never a hole;
     6→//!  - resume requests carry `If-Range` with the validator of the source the part came from;
     7→//!    a 200 answer means the source changed → the whole run restarts from zero (bounded);
     8→//!  - `206` answers are validated against the requested offset and total before any write;
     9→//!  - writes are hard-capped at the planned window end — server surplus bytes are dropped;
    10→//!  - completion verifies Σ written == total, then hashes the part file with SHA-256 and —
    11→//!    when the job carries an expected checksum — refuses to declare success on mismatch;
    12→//!  - the final rename happens only after fsync.
    13→//!
    14→//! Jobs whose source does not support ranges (or whose size is unknown) run as a single
    15→//! append-mode window; the model is identical, only the window is open-ended. A v1-era part
    16→//! (bytes on disk, validators persisted, no journal rows) is migrated forward instead of
    17→//! being discarded.
    18→

    90→#[derive(thiserror::Error, Debug)]
    91→pub enum EngineError {
    92→    #[error("contract: {0}")]
    93→    Contract(#[from] ContractError),
    94→    #[error("http: {0}")]
    95→    Http(#[from] reqwest::Error),
    96→    #[error("io: {0}")]
    97→    Io(#[from] std::io::Error),
    98→}
    99→
   100→impl EngineError {
   101→    fn to_contract(&self) -> ContractError {
   102→        match self {
   103→            EngineError::Contract(c) => c.clone(),
   104→            EngineError::Http(e) => {
   105→                let status = e.status().map(|s| s.as_u16());
   106→                let code = if e.is_timeout() || e.is_connect() {
   107→                    ErrorCode::NetworkError
   108→                } else if status.is_some() {
   109→                    match status {
   110→                        Some(416) => ErrorCode::RangeNotSatisfiable,
   111→                        _ => ErrorCode::HttpError,
   112→                    }
   113→                } else {
   114→                    ErrorCode::NetworkError
   115→                };
   116→                ContractError::new(code, format!("{e}"))
   117→                    .with_data(serde_json::json!({ "status": status }))
   118→            }
   119→            EngineError::Io(e) => {
   120→                let code = if e.kind() == std::io::ErrorKind::StorageFull {
   121→                    ErrorCode::DiskFull
   122→                } else {
   123→                    ErrorCode::IoError
   124→                };
   125→                ContractError::new(code, format!("{e}"))
   126→            }
   127→        }
   128→    }
   129→}

   131→pub fn now_ms() -> i64 {
   132→    std::time::SystemTime::now()
   133→        .duration_since(std::time::UNIX_EPOCH)
   134→        .map(|d| d.as_millis() as i64)
   135→        .unwrap_or(0)
   136→}
   137→
   138→pub fn touch(job: &mut DownloadJob) {
   139→    job.updated_at = now_ms();
   140→}
   141→
   142→fn set_error(job: &mut DownloadJob, err: &ContractError) {
   143→    job.status = "failed".into();
   144→    job.error_code = Some(err.code.as_str().into());
   145→    job.speed_bps = None;
   146→    job.eta_seconds = None;
   147→    touch(job);
   148→}

   150→/// Outcome of one segment task.
   151→#[derive(Debug, Clone, Copy, PartialEq)]
   152→enum SegmentOutcome {
   153→    /// The window is fully on disk.
   154→    Done,
   155→    /// The source changed under us (200 on a ranged request / total mismatch):
   156→    /// the caller wipes the journal and restarts the run.
   157→    Restart,
   158→    /// Supervisor teardown observed inside the task; the run must not continue.
   159→    Stopped,
   160→}
   161→
   162→/// Per-run shared state across the segment tasks.
   163→#[derive(Debug)]
   164→struct RunState {
   165→    /// Σ of every segment's in-memory write cursor (authoritative live received count).
   166→    received: Arc<AtomicU64>,
   167→    /// Set by the first task that detects a source change; peers exit early.
   168→    restart: Arc<AtomicBool>,
   169→    /// In-memory write cursors per segment index — flushed to the journal on pause/cancel
   170→    /// so a user-initiated pause resumes exactly where the UI says it did.
   171→    cursors: Arc<Mutex<Vec<u64>>>,
   172→}

   174→pub struct Engine {

   176→    pub client: reqwest::Client,

   181→impl Engine {
   182→    pub fn new(store: Arc<JobStore>, events: EventsTx) -> Self {
   183→        let client = reqwest::Client::builder()
   184→            .user_agent(concat!("UniDM/", env!("CARGO_PKG_VERSION")))
   185→            .connect_timeout(Duration::from_secs(20))
   186→            .build()
   187→            .expect("reqwest client builds");
   188→        Self {
   189→            store,
   190→            client,
   191→            events,
   192→            max_connections: Arc::new(AtomicU64::new(4)),
   193→        }
   194→    }
   195→
   196→    /// Connections per download (segmentation ceiling); clamped to the contract range.
   197→    pub fn set_max_connections(&self, max: usize) {
   198→        self.max_connections
   199→            .store(max.clamp(1, MAX_CONNECTIONS_CAP) as u64, Ordering::SeqCst);
   200→    }
   201→
   202→    pub fn emit_job(&self, job: &DownloadJob) {
   203→        let _ = self.events.send(EventEnvelope {
   204→            v: dc_protocol::PROTOCOL_VERSION,
   205→            event: "download.updated".into(),
   206→            payload: Some(serde_json::json!({ "job": job })),
   207→        });
   208→    }
   209→
   210→    pub fn emit_progress(&self, job: &DownloadJob) {
   211→        let _ = self.events.send(EventEnvelope {
   212→            v: dc_protocol::PROTOCOL_VERSION,
   213→            event: "download.progress".into(),
   214→            payload: Some(serde_json::json!({
   215→                "jobId": job.id.as_str(),
   216→                "receivedBytes": job.received_bytes,
   217→                "totalBytes": job.total_bytes,
   218→                "speedBps": job.speed_bps.unwrap_or(0),
   219→                "etaSeconds": job.eta_seconds,
   220→            })),
   221→        });
   222→    }
   223→
   224→    /// Create + persist a queued job; the destination dir is authoritative from creation.
   225→    pub async fn create_job(&self, params: CreateJobParams) -> Result<DownloadJob, ContractError> {
   226→        if !params.url.starts_with("http://") && !params.url.starts_with("https://") {
   227→            return Err(ContractError::new(
   228→                ErrorCode::UrlInvalid,
   229→                "only http(s) URLs are supported",
   230→            ));
   231→        }
   232→        let checksum = match &params.checksum {
   233→            Some(spec) => {
   234→                let value = normalize_expected(spec.algorithm, &spec.value)
   235→                    .map_err(|msg| ContractError::new(ErrorCode::InvalidParams, msg))?;
   236→                Some((spec.algorithm, value))
   237→            }
   238→            None => None,
   239→        };
   240→        let ts = now_ms();
   241→        let job = DownloadJob {
   242→            id: new_ulid(),
   243→            url: params.url,
   244→            destination_dir: params.destination_dir,
   245→            destination_path: None, // resolved by the probe
   246→            filename: params.filename,
   247→            status: "queued".into(),
   248→            total_bytes: None,
   249→            received_bytes: 0,
   250→            etag: None,
   251→            last_modified: None,
   252→            content_type: None,
   253→            speed_bps: None,
   254→            eta_seconds: None,
   255→            error_code: None,
   256→            headers: params.headers.clone(),
   257→            segment_count: None,
   258→            checksum_algorithm: checksum.as_ref().map(|(a, _)| *a),
   259→            checksum_expected: checksum.as_ref().map(|(_, v)| v.clone()),
   260→            checksum_actual: None,
   261→            checksum_verified: None,
   262→            created_at: ts,
   263→            updated_at: ts,
   264→        };
   265→        self.store.insert_job(&job)?;
   266→        Ok(job)
   267→    }
   268→
   269→    /// The per-job task: probe → segmented download → verify. Runs until terminal/paused.
   270→    pub async fn run_job(
   271→        self: Arc<Self>,
   272→        mut job: DownloadJob,
   273→        mut cmd_rx: mpsc::Receiver<JobCommand>,
   274→        stop: CancellationToken,
   275→    ) {
   276→        match self.drive(&mut job, &mut cmd_rx, &stop).await {
   277→            Ok(()) => {}
   278→            Err(err) => {
   279→                if !stop.is_cancelled() {
   280→                    let cerr = err.to_contract();
   281→                    tracing::warn!(job = %job.id, error = %cerr, "job failed");
   282→                    set_error(&mut job, &cerr);
   283→                    if self.store.update_job(&job).is_ok() {
   284→                        self.emit_job(&job);
   285→                    }
   286→                }
   287→            }
   288→        }
   289→    }
   290→
   291→    async fn drive(
   292→        &self,
   293→        job: &mut DownloadJob,
   294→        cmd_rx: &mut mpsc::Receiver<JobCommand>,
   295→        stop: &CancellationToken,
   296→    ) -> Result<(), EngineError> {
   297→        // Resume safety: If-Range must carry the validators of the SOURCE THE PART CAME FROM
   298→        // (persisted before this run), never the freshly probed ones — otherwise a changed
   299→        // source would answer 206 against the new validator and the part would be spliced
   300→        // with foreign bytes. Captured BEFORE the probe overwrites job.etag/last_modified.
   301→        let mut part_etag = job.etag.clone();
   302→        let mut part_last_modified = job.last_modified.clone();
   303→
   304→        // ---- probing ----
   305→        job.status = "probing".into();
   306→        touch(job);
   307→        self.store.update_job(job)?;
   308→        self.emit_job(job);
   309→
   310→        let mut probe = tokio::select! {
   311→            _ = stop.cancelled() => return Ok(()),
   312→            r = self.probe(job) => r?,
   313→        };
   314→
   315→        // ---- destination resolution (once; the part file is bound to this path) ----
   316→        let base_name = job
   317→            .filename
   318→            .clone()
   319→            .or_else(|| probe.filename_hint.clone())
   320→            .or_else(|| filename_from_url(&job.url))
   321→            .unwrap_or_else(|| "download.bin".into());
   322→        let dir = PathBuf::from(&job.destination_dir);
   323→        tokio::fs::create_dir_all(&dir)
   324→            .await
   325→            .map_err(EngineError::Io)?;
   326→        let candidate = dir.join(sanitize_filename(&base_name));
   327→        let final_path = if tokio::fs::try_exists(&candidate).await.unwrap_or(false) {
   328→            uniquify(&candidate).await
   329→        } else {
   330→            candidate
   331→        };
   332→        let part_path = part_file(&final_path);
   333→
   334→        job.destination_path = Some(final_path.display().to_string());
   335→        job.total_bytes = probe.total_bytes;
   336→        job.etag = probe.etag.clone();
   337→        job.last_modified = probe.last_modified.clone();
   338→        job.content_type = probe.content_type.clone();
   339→        touch(job);
   340→        self.store.update_job(job)?;
   341→        self.emit_job(job);
   342→
   343→        // ---- segmented run loop (bounded clean restarts on source change) ----
   344→        let mut restarts_used = 0usize;
   345→        loop {
   346→            let max_conn = self.max_connections.load(Ordering::SeqCst) as usize;
   347→            let windows = plan_windows(
   348→                job.total_bytes,
   349→                probe.ranges_supported,
   350→                max_conn,
   351→                MIN_SEGMENT_BYTES,
   352→            );
   353→
   354→            // Journal reconciliation. A v1-era part (no journal rows) is migrated forward
   355→            // instead of thrown away; a plan change resets the journal.
   356→            let mut journal = self.store.list_segments(job.id.as_str())?;
   357→            if journal.is_empty() {
   358→                journal = self.migrate_v1_part(job, &part_path, &windows).await?;
   359→            }
   360→            if !journal_matches_plan(&journal, &windows) {
   361→                tracing::info!(job = %job.id, "plan changed — resetting segment journal");
   362→                journal = journal_from_plan(&windows);
   363→                self.store
   364→                    .replace_segments(job.id.as_str(), &journal)
   365→                    .map_err(EngineError::Contract)?;
   366→            }
   367→            // A part with bytes but no validators cannot be proven to match the source:
   368→            // integrity policy discards it rather than risking a hybrid file.
   369→            let part_validators = part_etag.clone().or_else(|| part_last_modified.clone());
   370→            let has_written = journal.iter().any(|s| s.written_bytes > 0);
   371→            if has_written && part_validators.is_none() {
   372→                tracing::warn!(job = %job.id, "unprovable part (no validators) — discarding");
   373→                journal = journal_from_plan(&windows);
   374→                self.store
   375→                    .replace_segments(job.id.as_str(), &journal)
   376→                    .map_err(EngineError::Contract)?;
   377→            }
   378→
   379→            let offset_mode = windows.iter().all(|w| w.end.is_some());
   380→            let initial_written: u64 = journal.iter().map(|s| s.written_bytes).sum();
   381→
   382→            // Align the part file with the journal before any writer opens it.
   383→            if offset_mode {
   384→                // Sparse pre-size: the windows tile [0, total) exactly.
   385→                let file = OpenOptions::new()
   386→                    .create(true)
   387→                    .truncate(false)
   388→                    .write(true)
   389→                    .open(&part_path)
   390→                    .await
   391→                    .map_err(EngineError::Io)?;
   392→                file.set_len(job.total_bytes.unwrap_or(0))
   393→                    .await
   394→                    .map_err(EngineError::Io)?;
   395→            } else {
   396→                // Append mode: truncate to the durable cursor (drops any stale crash tail).
   397→                let file = OpenOptions::new()
   398→                    .create(true)
   399→                    .truncate(false)
   400→                    .write(true)
   401→                    .open(&part_path)
   402→                    .await
   403→                    .map_err(EngineError::Io)?;
   404→                file.set_len(initial_written)
   405→                    .await
   406→                    .map_err(EngineError::Io)?;
   407→            }
   408→
   409→            // The If-Range validator for every request of this run: the PART's validators
   410→            // when resuming written data, otherwise the freshly probed ones.
   411→            let if_range = if has_written {
   412→                part_validators.clone()
   413→            } else {
   414→                job.etag.clone().or_else(|| job.last_modified.clone())
   415→            };
   416→
   417→            job.segment_count = Some(windows.len().max(1) as u64);
   418→            job.status = "downloading".into();
   419→            job.received_bytes = initial_written;
   420→            job.speed_bps = None;
   421→            touch(job);
   422→            self.store.update_job(job)?;
   423→            self.emit_job(job);
   424→
   425→            let state = Arc::new(RunState {
   426→                received: Arc::new(AtomicU64::new(initial_written)),
   427→                restart: Arc::new(AtomicBool::new(false)),
   428→                cursors: Arc::new(Mutex::new(
   429→                    journal.iter().map(|s| s.written_bytes).collect(),
   430→                )),
   431→            });
   432→
   433→            // Progress ticker: one throttled stream of updates for the whole run.
   434→            let ticker_stop = CancellationToken::new();
   435→            let ticker = self.spawn_ticker(
   436→                job.id.as_str().to_string(),
   437→                job.total_bytes,
   438→                state.received.clone(),
   439→                ticker_stop.clone(),
   440→            );
   441→
   442→            // One task per window.
   443→            let mut tasks: JoinSet<Result<SegmentOutcome, EngineError>> = JoinSet::new();
   444→            for (idx, window) in windows.iter().enumerate() {
   445→                let row = journal[idx].clone();
   446→                tasks.spawn(Self::run_segment(
   447→                    self.store.clone(),
   448→                    self.client.clone(),
   449→                    job.clone(),
   450→                    idx,
   451→                    *window,
   452→                    row,
   453→                    part_path.clone(),
   454→                    if_range.clone(),
   455→                    state.clone(),
   456→                    stop.clone(),
   457→                ));
   458→            }
   459→
   460→            let mut run_result: Result<(), EngineError> = Ok(());
   461→            let mut source_changed = false;
   462→            while !tasks.is_empty() {
   463→                tokio::select! {
   464→                    _ = stop.cancelled() => {
   465→                        tasks.abort_all();
   466→                        ticker_stop.cancel();
   467→                        let _ = ticker.await;
   468→                        return Ok(());
   469→                    }
   470→                    cmd = cmd_rx.recv() => {
   471→                        tasks.abort_all();
   472→                        ticker_stop.cancel();
   473→                        let _ = ticker.await;
   474→                        self.finalize_run(job, &state, "commanded stop");
   475→                        if let Some(cmd) = cmd {
   476→                            match cmd {
   477→                                JobCommand::Pause => {
   478→                                    job.status = "paused".into();
   479→                                    job.speed_bps = None;
   480→                                    job.eta_seconds = None;
   481→                                    touch(job);
   482→                                    self.store.update_job(job)?;
   483→                                    self.emit_job(job);
   484→                                }
   485→                                JobCommand::Cancel => {
   486→                                    job.status = "canceled".into();
   487→                                    job.speed_bps = None;
   488→                                    job.eta_seconds = None;
   489→                                    touch(job);
   490→                                    self.store.update_job(job)?;
   491→                                    self.emit_job(job);
   492→                                }
   493→                            }
   494→                        }
   495→                        return Ok(());
   496→                    }
   497→                    done = tasks.join_next(), if !tasks.is_empty() => {
   498→                        match done {
   499→                            Some(Ok(Ok(SegmentOutcome::Done))) => {}
   500→                            Some(Ok(Ok(SegmentOutcome::Stopped))) => {
   501→                                tasks.abort_all();
   502→                                ticker_stop.cancel();
   503→                                let _ = ticker.await;
   504→                                return Ok(());
   505→                            }
   506→                            Some(Ok(Ok(SegmentOutcome::Restart))) => {
   507→                                source_changed = true;
   508→                                tasks.abort_all();
   509→                            }
   510→                            Some(Ok(Err(err))) => {
   511→                                run_result = Err(err);
   512→                                tasks.abort_all();
   513→                            }
   514→                            Some(Err(join)) => {
   515→                                run_result = Err(EngineError::Contract(ContractError::new(
   516→                                    ErrorCode::InternalError,
   517→                                    format!("segment task panicked: {join}"),
   518→                                )));
   519→                                tasks.abort_all();
   520→                            }
   521→                            None => {}
   522→                        }
   523→                    }
   524→                }
   525→            }
   526→            ticker_stop.cancel();
   527→            let _ = ticker.await;
   528→            if stop.is_cancelled() {
   529→                return Ok(());
   530→            }
   531→
   532→            if let Err(err) = run_result {
   533→                self.finalize_run(job, &state, "failure");
   534→                return Err(err);
   535→            }
   536→
   537→            if source_changed {
   538→                if restarts_used >= MAX_SOURCE_RESTARTS {
   539→                    return Err(EngineError::Contract(ContractError::new(
   540→                        ErrorCode::HttpError,
   541→                        "source changed repeatedly — giving up after clean restarts",
   542→                    )));
   543→                }
   544→                restarts_used += 1;
   545→                tracing::info!(job = %job.id, restart = restarts_used, "source changed — clean restart");
   546→
   547→                // Wipe the journal and the part BEFORE anything can re-migrate the stale
   548→                // bytes, then re-probe so the next pass carries fresh validators.
   549→                state.received.store(0, Ordering::SeqCst);
   550→                self.store.delete_segments(job.id.as_str())?;
   551→                let file = OpenOptions::new()
   552→                    .create(true)
   553→                    .truncate(false)
   554→                    .write(true)
   555→                    .open(&part_path)
   556→                    .await
   557→                    .map_err(EngineError::Io)?;
   558→                file.set_len(0).await.map_err(EngineError::Io)?;
   559→                part_etag = None;
   560→                part_last_modified = None;
   561→
   562→                job.status = "probing".into();
   563→                touch(job);
